function voltage_variance ()
    grid_logic_var.RANDVOLTAGE = math.random()
    if (grid_logic_var.RANDVOLTAGE > 0.500090439276485) then
        grid_logic_var.VOLTPLUS = "+"
    else
        grid_logic_var.VOLTPLUS = "-"
    end
    grid_logic_var.VOLTVALUE = (math.floor((36000*grid_logic_var.RANDVOLTAGE*(math.min(grid_logic_var.MIN_P_POWER,grid_logic_var.ALL_P_POWER)/math.max(grid_logic_var.ALL_P_POWER,0.1)))-(18000*(math.min(grid_logic_var.MIN_P_POWER,grid_logic_var.ALL_P_POWER)/math.max(grid_logic_var.ALL_P_POWER,0.1))))/1000)
    return grid_logic_var.VOLTPLUS..grid_logic_var.VOLTVALUE.." volts"
end

-- Flavour values of the query windows (Stuck Mirror Motors, Bird Carcasses, ...).
-- Uses game.cam_get_<name>() from CAM_LUA_Controller.dat; if CAM is missing or the function gives nil,
-- a random value 0..max, so the window never shows the raw "#math.random()#" caption.
function neon_cam_value(name, max)
    local f = game["cam_get_" .. name]
    if (type(f) == "function") then
        local v = f()
        if (v ~= nil) then return v end
    end
    return math.floor(math.random() * (max + 1))
end

function run_grid_balance_variables(ret_type)
    if (neon.update == nil) then
        -- SC4NEON.dll is not installed: the grid values stay at 0 and the rewards stay locked.
        if (ret_type ~= nil) then return 0 end
        return
    end
-- Determine if a bridge is present (unchanged from the Lua version)
    if (game.g_bridge_tile_count < unlock_variables.TRANS_SUBSTATION_60_MVA_S_BRIDGE_TILES) then
        unlock_variables.TRANS_SUBSTATION_60_MVA_S_BRIDGE = "No"
    else
        unlock_variables.TRANS_SUBSTATION_60_MVA_S_BRIDGE = "Yes"
    end
    local level = 2
    if (game.difficulty_level() == game_difficulty_level.EASY) then
        level = 0
    elseif (game.difficulty_level() == game_difficulty_level.MEDIUM) then
        level = 1
    end
    if (neon.update(game.trend_value(game_trends.G_POWER_DEMANDED,0),
                    game.trend_value(game_trends.G_POWER_EXPORTED,0),
                    game.trend_value(game_trends.G_POWER_IMPORTED,0),
                    game.trend_value(game_trends.G_POWER_PRODUCED,0),
                    game.trend_value(game_trends.G_GARBAGE_TO_ENERGY,0),
                    game.g_power_consumed,
                    game.g_power_production_capacity,
                    level,
                    game.g_city_r_population)) then
        neon.export(grid_logic_var, power_company_buildings, kva_rating, mwh_rating)

    end

        return neon.grid(ret_type) or 0
